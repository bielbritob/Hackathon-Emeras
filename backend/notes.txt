- Obrigatório
    cd backend
    python3 -m venv venv
    source venv/bin/activate

- Baixar requirements
    pip3 install -r requirements.txt

- Atualizar requirements
    pip3 freeze > requirements.txt

-- Rodar
    fastapi dev src/main.py
    uvicorn src.main:app --reload

-- Pytest
    pytest -v #verbose
